# 1. Gunakan image Python versi ringan (slim) untuk mempercepat proses build
FROM python:3.10-slim

# 2. Tentukan direktori kerja di dalam container
WORKDIR /app

# 3. Salin file requirements.txt ke dalam container
COPY requirements.txt .

# 4. Instal library yang dibutuhkan tanpa menyimpan cache (agar ukuran image kecil)
RUN pip install --no-cache-dir -r requirements.txt

# 5. Salin seluruh kode aplikasi Anda (NewsIntel.py) ke dalam container
COPY . .

# 6. Ekspos port 5000 (Sesuai dengan port bawaan Flask di skrip Anda)
EXPOSE 5000

# 7. Jalankan aplikasi menggunakan Gunicorn. 
# Perhatikan "NewsIntel:app" karena nama file Anda adalah NewsIntel.py
CMD ["gunicorn", "NewsIntel:app", "--bind", "0.0.0.0:5000"]
